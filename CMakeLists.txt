# gpplus - gamepad input for C++ without a platform layer attached.
#
# Consume it with FetchContent:
#
#   include(FetchContent)
#   FetchContent_Declare(gpplus
#     GIT_REPOSITORY https://github.com/SeedyROM/gpplus.git
#     GIT_TAG        v0.1.0-alpha)
#   FetchContent_MakeAvailable(gpplus)
#   target_link_libraries(mygame PRIVATE gpplus::gpplus)
#
# or with find_package(gpplus) after an install. Nothing else is required:
# no submodules, no package manager, no second platform layer beside the
# one your game already has.

cmake_minimum_required(VERSION 3.16)
project(gpplus VERSION 0.1.1 LANGUAGES CXX)

if(APPLE)
  # Needed before any Objective-C++ probe runs, and harmless otherwise.
  enable_language(OBJCXX)
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")

# Sub-project builds should not bring tests, examples or install rules with
# them; a game that FetchContents this wants a library and nothing else.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(gpplus_is_top_level TRUE)
else()
  set(gpplus_is_top_level FALSE)
endif()

option(GPPLUS_BUILD_EXAMPLES "Build the example programs" ${gpplus_is_top_level})
option(GPPLUS_BUILD_TESTS "Build the tests" ${gpplus_is_top_level})
option(GPPLUS_INSTALL "Generate install rules" ${gpplus_is_top_level})
option(GPPLUS_BUNDLE_CONTROLLER_DB
       "Compile the controller mapping database into the library" ON)
option(GPPLUS_DOWNLOAD_CONTROLLER_DB
       "Fetch the mapping database at configure time if it is not already here" ON)
option(GPPLUS_BACKEND_VIRTUAL
       "Build the synthetic backend (gpplus/virtual.hpp) for tests and development without hardware" ${GPPLUS_BUILD_TESTS})
option(GPPLUS_WERROR "Treat the library's own warnings as errors" OFF)
set(GPPLUS_CONTROLLER_DB_FILE "" CACHE FILEPATH
    "Use this mapping database instead of downloading one")

include(GpplusBackends)
include(GpplusControllerDB)

gpplus_backend_option(GAMECONTROLLER "Apple GameController.framework backend")
gpplus_backend_option(IOKIT "Apple raw HID (IOKit) backend")
gpplus_backend_option(EVDEV "Linux evdev backend")
gpplus_backend_option(XINPUT "Windows XInput backend")
gpplus_backend_option(DINPUT "Windows DirectInput 8 backend, for non-XInput hardware")

gpplus_detect_backends()

if(GPPLUS_BACKEND_IDS)
  message(STATUS "gpplus: backends: ${GPPLUS_BACKEND_IDS}")
else()
  # Not an error. A build for a platform with no backend is a working
  # build that reports no gamepads, and a game that links it still runs --
  # which beats failing to configure on someone's BSD.
  message(WARNING
    "gpplus: no backend was detected for this platform. The library will "
    "build and report no devices.")
endif()

# The backend list, as a file the sources include twice with different
# definitions of GPPLUS_BACKEND -- once to declare the factories, once to
# call them. Adding a platform means adding a file and a detection block,
# and touching no existing code.
set(backends_inc_body "")
foreach(id IN LISTS GPPLUS_BACKEND_IDS)
  string(APPEND backends_inc_body "GPPLUS_BACKEND(${id})\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/gpplus_backends.inc"
     CONTENT "// Generated by CMake. One line per backend compiled in.\n${backends_inc_body}"
     @ONLY)

set(GPPLUS_BACKEND_DEFINES "")
foreach(id IN LISTS GPPLUS_BACKEND_IDS)
  string(TOUPPER "${id}" upper)
  string(APPEND GPPLUS_BACKEND_DEFINES "#define GPPLUS_BACKEND_${upper} 1\n")
endforeach()

gpplus_prepare_controller_db(GPPLUS_DB_SOURCE)
if(GPPLUS_CONTROLLER_DB_ENTRIES AND GPPLUS_CONTROLLER_DB_ENTRIES GREATER 0)
  set(GPPLUS_HAS_BUILTIN_DB 1)
  set(GPPLUS_BUILTIN_DB_ENTRIES ${GPPLUS_CONTROLLER_DB_ENTRIES})
else()
  set(GPPLUS_HAS_BUILTIN_DB 0)
  set(GPPLUS_BUILTIN_DB_ENTRIES 0)
endif()

_gpplus_db_platform_tag(GPPLUS_PLATFORM_TAG)
configure_file("${CMAKE_CURRENT_LIST_DIR}/cmake/config.hpp.in"
               "${CMAKE_CURRENT_BINARY_DIR}/generated/gpplus/config.hpp" @ONLY)

add_library(gpplus
  src/binding.cpp
  src/context.cpp
  src/mapping.cpp
  "${GPPLUS_DB_SOURCE}"
  ${GPPLUS_BACKEND_SOURCES})
add_library(gpplus::gpplus ALIAS gpplus)

target_compile_features(gpplus PUBLIC cxx_std_17)
target_include_directories(gpplus
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>
    $<INSTALL_INTERFACE:include>
  PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(gpplus PRIVATE ${GPPLUS_BACKEND_LIBS})
set_target_properties(gpplus PROPERTIES
  VERSION ${PROJECT_VERSION}
  SOVERSION ${PROJECT_VERSION_MAJOR}
  POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON)

if(APPLE)
  # ARC for the Objective-C++ backends. They hold framework objects across
  # frames -- a haptic engine, a controller -- and manual retain/release on
  # those is exactly the sort of thing that works until a pad is unplugged
  # at the wrong moment.
  set_source_files_properties(${GPPLUS_BACKEND_SOURCES} PROPERTIES
    COMPILE_OPTIONS "-fobjc-arc")
endif()

if(MSVC)
  target_compile_options(gpplus PRIVATE /W4 $<$<BOOL:${GPPLUS_WERROR}>:/WX>)
  # std::getenv is standard C++, and MSVC's C4996 nags about it (with /WX,
  # fails the build) in favour of _dupenv_s, which exists nowhere else.
  # Portable code is the point, so the warning is what goes.
  target_compile_definitions(gpplus PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(gpplus PRIVATE
    -Wall -Wextra -Wpedantic $<$<BOOL:${GPPLUS_WERROR}>:-Werror>)
endif()

if(GPPLUS_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()

if(GPPLUS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

if(GPPLUS_INSTALL)
  include(GNUInstallDirs)
  include(CMakePackageConfigHelpers)

  install(TARGETS gpplus EXPORT gpplusTargets
          RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
          LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
          ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/generated/gpplus/config.hpp"
          DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/gpplus)
  install(FILES LICENSE
          DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/gpplus)
  install(EXPORT gpplusTargets NAMESPACE gpplus::
          DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/gpplus)

  configure_package_config_file(cmake/gpplusConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/gpplusConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/gpplus)
  write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/gpplusConfigVersion.cmake"
    COMPATIBILITY SameMajorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/gpplusConfig.cmake"
                "${CMAKE_CURRENT_BINARY_DIR}/gpplusConfigVersion.cmake"
          DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/gpplus)
endif()
